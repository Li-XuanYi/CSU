def plot_windy_world(rows, cols, wind_rows=None, wind_strength=1, paths=None):
    """绘制 Windy World 网格图。

    Args:
        rows: 网格行数
        cols: 网格列数
        wind_rows: 有风的行索引列表（默认第1行）
        wind_strength: 风力大小（默认1）
        dp: count_paths 返回的路径数表，若为 None 则不显示数字
    """
    start = (0, 0)
    goal  = (rows - 1, cols - 1)

    # 风力矩阵
    s_wind = np.zeros((rows, cols), dtype=int)
    for wr in wind_rows:
        s_wind[wr, :] = wind_strength

    fig, ax = plt.subplots(figsize=(7, 5))

    # 底图颜色
    wind_colors = {
        0: [0.95, 0.95, 0.95],
        1: [0.65, 0.82, 0.95],
    }
    grid = np.ones((rows, cols, 3))
    for r in range(rows):
        for c in range(cols):
            grid[r, c] = wind_colors[s_wind[r, c]]

    grid[start] = [0.4,  0.73, 0.42]
    grid[goal]  = [0.08, 0.33, 0.73]

    ax.imshow(grid, origin='upper')

    # 网格分界线
    for r in range(rows + 1):
        ax.axhline(r - 0.5, color='#888', linewidth=0.8)
    for c in range(cols + 1):
        ax.axvline(c - 0.5, color='#888', linewidth=0.8)

    # 图例
    legend_elements = [
        Patch(facecolor=[0.65, 0.82, 0.95], label=f'Wind →{wind_strength}'),
        Patch(facecolor=[0.95, 0.95, 0.95], label='No wind'),
        Patch(facecolor=[0.4,  0.73, 0.42], label='Start S'),
        Patch(facecolor=[0.08, 0.33, 0.73], label='Goal G'),
    ]
    ax.legend(handles=legend_elements, loc='upper left', fontsize=9,
              bbox_to_anchor=(1.02, 1), borderaxespad=0)

    ax.set_xticks(range(cols))
    ax.set_yticks(range(rows))

    # dp 路径数：显示在每个格子中央
    if paths is not None:
        for r in range(rows):
            for c in range(cols):
                val = paths[r][c]
                txt_color = '#1a5276' if s_wind[r, c] > 0 else '#333333'
                ax.text(c, r, str(val), ha='center', va='center',
                        fontsize=13, fontweight='bold', color=txt_color)

    # 右下角小字：风力标注 + 起点/终点标签
    for r in range(rows):
        for c in range(cols):
            annotations = []
            w = s_wind[r, c]
            if w > 0:
                annotations.append(f'→{w}')
            if (r, c) == start:
                annotations.append('S')
            if (r, c) == goal:
                annotations.append('G')
            if annotations:
                ax.text(c + 0.38, r + 0.38, ' '.join(annotations),
                        ha='right', va='bottom', fontsize=12, color='#FFF')

    ax.set_title(f'Windy World  (S={start}  G={goal})', fontsize=12)

    plt.tight_layout()
    plt.savefig('Fig-Windy World.png', dpi=150, bbox_inches='tight')
    plt.show()

def recursive_unique_paths(m,n,w,s,i=0,j=0):
    if i>=m or j>=n: return 0
    if i==m-1 and j==n-1: return 1
    # 向下走
    di,dj=i+1,j
    if i<m-1 and di in w: dj=min(dj+s,n-1)
    # 向右走
    ri,rj=i,j+1
    if j<n-1 and i in w: rj=min(rj+s,n-1)
    return recursive_unique_paths(m,n,w,s,di,dj)+recursive_unique_paths(m,n,w,s,ri,rj)

def memoization_unique_paths(m,n,w,s):
    dp=[[-1]*n for _ in range(m)]
    def _dp(i,j):
        if i>=m or j>=n: return 0
        if i==m-1 and j==n-1: return 1
        if dp[i][j]!=-1: return dp[i][j]
        # 向下走
        di,dj=i+1,j
        if i<m-1 and di in w: dj=min(dj+s,n-1)
        # 向右走
        ri,rj=i,j+1
        if j<n-1 and i in w: rj=min(rj+s,n-1)
        dp[i][j]=_dp(di,dj)+_dp(ri,rj)
        return dp[i][j]
    return _dp(0,0)

def tabulation_unique_paths(m, n, w, s):
    dp = [[0]*n for _ in range(m)]
    dp[0][0] = 1
    for i in range(m):
        for j in range(n):
            if dp[i][j]:
                if j < n-1: # 往右
                    nj = j+1 + (s if i in w else 0)
                    dp[i][min(nj, n-1)] += dp[i][j]
                if i < m-1: # 往下
                    nj = j + (s if i+1 in w else 0)
                    dp[i+1][min(nj, n-1)] += dp[i][j]
    return dp[m-1][n-1], dp
