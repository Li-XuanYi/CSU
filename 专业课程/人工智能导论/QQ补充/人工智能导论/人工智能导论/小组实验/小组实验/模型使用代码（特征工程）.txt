import torch
import pandas as pd
import numpy as np
from torch import nn
from sklearn.preprocessing import StandardScaler
from joblib import load  # 用于加载保存的标准化器

# --------------------------
# 1. 定义模型结构（必须与训练时完全一致）
# --------------------------
class LogisticRegression(nn.Module):
    def __init__(self, input_dim):
        super().__init__()
        self.linear = nn.Linear(input_dim, 1)
        self.sigmoid = nn.Sigmoid()
        
    def forward(self, x):
        return self.sigmoid(self.linear(x))

# --------------------------
# 2. 加载模型和标准化器
# --------------------------
def load_resources(model_path, scaler_path):
    # 加载标准化器（假设训练时已保存）
    scaler = load(scaler_path) 
    
    # 初始化模型
    model = LogisticRegression(input_dim=18)  # 替换为实际特征数量
    model.load_state_dict(torch.load(model_path))
    model.eval()  # 设置为评估模式
    
    return model, scaler

# 3. 数据预处理流水线
# --------------------------
num_columns=['年龄','em','1ET天','胚芽','卵黄囊','胎心']

def map_iii_line(value):
    if pd.isna(value):
        return 'Ⅲ线_欠清'
    value = str(value).strip()
    if value in {'微量宫腔积液', '少量宫腔积液', '宫腔积液', '积液', '微量积液', '宫腔积水', '大量宫腔积液', '少量积液'}:
        return 'Ⅲ线_有积液'
    elif value in {'1'}:
        return 'Ⅲ线_Ⅰ'
    elif value in {'2稍欠清', '2                2', '2', '2均'}:
        return 'Ⅲ线_Ⅱ'
    elif value in {'3                3', '3均', '3均匀', '33', '3均，宫腔积液', '3'}:
        return 'Ⅲ线_Ⅲ'
    elif value in {'2-3'}:
        return 'Ⅲ线_Ⅱ-Ⅲ'
    else:
        return 'Ⅲ线_欠清'

        processed_data_0['孕囊'].astype(str)

        .str.replace('x', 'X', case=False)  # 统一为X分隔符

    )

        processed_data_0['孕囊'].str.split('X', expand=True, n=1)  # 最多分割1次得到两列

        'Ⅲ线_Ⅲ':     [0, 0, 0, 1, 0, 0],

    # 合并到原数据中
    processed_data_0 = pd.concat([processed_data_0, onehots], axis=1)

    # 6. 丢弃原始“Ⅲ线”和“助孕方式”列（可选）

    feature_data_0=processed_data_0[['年龄', 'em','1ET天',     
    '胚芽',

    new_data_1 = pd.read_excel(data_1_path, engine="openpyxl")  # 支持CSV/Excel等格式

        .str.replace('x', 'X', case=False)  # 统一为X分隔符

# --------------------------
# 4. 执行预测
# --------------------------
def predict(model, scaled_data, threshold=0.5):
    tensor_data = torch.tensor(scaled_data, dtype=torch.float32)
    
    with torch.no_grad():
        probabilities = model(tensor_data).numpy().flatten()
    
    predictions = (probabilities > threshold).astype(int)
    return probabilities, predictions
